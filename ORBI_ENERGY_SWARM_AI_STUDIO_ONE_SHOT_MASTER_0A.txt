import React, { useEffect, useRef, useState, useCallback } from "react";
import { 
  Shield, Award, Coins, Zap, Play, RotateCcw, Volume2, VolumeX, 
  Pause, Users, Database, BookOpen, Sparkles, Compass, Cpu
} from "lucide-react";

import { 
  BiomeType, FormationType, OrbiType, EnemyType, 
  OrbiMember, Enemy, Projectile, Resource, Particle, GameStats, DialogueMessage 
} from "./types";

import { 
  CANVAS_WIDTH, CANVAS_HEIGHT, PLAYER_SPEED, PLAYER_MAX_SHIELD, 
  PLAYER_SHIELD_REGEN, DAMAGE_INVINCIBILITY_DURATION,
  MAX_SWARM_MEMBERS, MAX_ENEMIES, MAX_PROJECTILES, MAX_RESOURCES, MAX_PARTICLES,
  BIOMES, FORMATIONS, ORBI_TEMPLATES, COMPANION_QUOTES
} from "./constants";

import { loadGameStats, saveGameStats } from "./storage";

import { 
  playShootSound, playEnemyShootSound, playDamageSound, playEnemyExplodeSound, 
  playRecruitSound, playCrystalSound, playBiomeChangeSound, playClickSound, 
  playPauseSound, playUnpauseSound, playGameOverSound, setMuteState, initAudio 
} from "./audio";

export const EnergySwarmGame: React.FC = () => {
  // --- REACT OVERLAY STATES (Updated at a throttled frequency) ---
  const [activeTab, setActiveTab] = useState<"swarm" | "civ" | "mementos" | "relics">("swarm");
  const [hudShield, setHudShield] = useState(100);
  const [hudScore, setHudScore] = useState(0);
  const [hudHighScore, setHudHighScore] = useState(0);
  const [hudPopulation, setHudPopulation] = useState(1);
  const [hudNano, setHudNano] = useState(0);
  const [hudEnergy, setHudEnergy] = useState(0);
  const [hudBiome, setHudBiome] = useState<BiomeType>(BiomeType.SOLAR_PLAINS);
  const [hudFormation, setHudFormation] = useState<FormationType>(FormationType.CIRCLE);
  const [isPaused, setIsPaused] = useState(false);
  const [isGameOver, setIsGameOver] = useState(false);
  const [isGameStarted, setIsGameStarted] = useState(false);
  const [muted, setMuted] = useState(false);
  const [dialogueLog, setDialogueLog] = useState<DialogueMessage[]>([]);
  const [stats, setStats] = useState<GameStats | null>(null);

  // Relic Upgrade State
  const [unlockedRelics, setUnlockedRelics] = useState<string[]>([]);

  // Companion avatar animations
  const [companionDialogue, setCompanionDialogue] = useState<string>(
    "Welcome Foton! Tap START core synchronization."
  );

  // --- CANVAS & STATE REFS (High-Frequency updates) ---
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const containerRef = useRef<HTMLDivElement | null>(null);

  const playerRef = useRef({
    x: CANVAS_WIDTH / 2,
    y: CANVAS_HEIGHT / 2,
    shield: PLAYER_MAX_SHIELD,
    maxShield: PLAYER_MAX_SHIELD,
    invincibilityTimer: 0,
    speed: PLAYER_SPEED,
    radius: 12
  });

  const pointerRef = useRef({ x: CANVAS_WIDTH / 2, y: CANVAS_HEIGHT / 2 });
  const usingKeyboardRef = useRef(false);
  const keysPressedRef = useRef<Record<string, boolean>>({});

  const swarmRef = useRef<OrbiMember[]>([]);
  const enemiesRef = useRef<Enemy[]>([]);
  const projectilesRef = useRef<Projectile[]>([]);
  const resourcesRef = useRef<Resource[]>([]);
  const particlesRef = useRef<Particle[]>([]);

  const scoreRef = useRef(0);
  const nanoCreditsRef = useRef(0);
  const purifiedEnergyRef = useRef(0);
  const activeBiomeRef = useRef<BiomeType>(BiomeType.SOLAR_PLAINS);
  const activeFormationRef = useRef<FormationType>(FormationType.CIRCLE);
  const isPausedRef = useRef(false);
  const isGameOverRef = useRef(false);
  const isGameStartedRef = useRef(false);
  
  const frameCountRef = useRef(0);
  const loopIdRef = useRef<number | null>(null);
  const lastTimeRef = useRef<number>(0);
  const lastDialogueTimeRef = useRef<number>(0);
  const biomeProgressRef = useRef(0); // Progress to next biome

  // Grid offsets for scrolling grid visual effect
  const gridOffsetRef = useRef({ x: 0, y: 0 });

  // --- COORDINATE MAPPING ---
  const handleCanvasPointer = (e: React.PointerEvent<HTMLCanvasElement>) => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const rect = canvas.getBoundingClientRect();
    const x = ((e.clientX - rect.left) / rect.width) * CANVAS_WIDTH;
    const y = ((e.clientY - rect.top) / rect.height) * CANVAS_HEIGHT;
    pointerRef.current = { x, y };
  };

  // --- COMPANION LOGIC ---
  const postDialogue = useCallback((sender: "FOTON" | "COMPANION" | "SYSTEM", text: string) => {
    const timestamp = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });
    const newMessage: DialogueMessage = {
      id: Math.random().toString(),
      sender,
      text,
      timestamp
    };
    setDialogueLog(prev => [newMessage, ...prev].slice(0, 30));
    if (sender === "COMPANION") {
      setCompanionDialogue(text);
    }
  }, []);

  // --- LOAD GAME STATS ---
  useEffect(() => {
    const loaded = loadGameStats();
    setStats(loaded);
    setHudHighScore(loaded.highScore);
    setMuted(loaded.audioMuted);
    setMuteState(loaded.audioMuted);
    
    // Load unlocked relics from saved memories/relics system
    const relics: string[] = [];
    loaded.gameMemories.forEach(mem => {
      if (mem.startsWith("Relic Unlocked: ")) {
        relics.push(mem.replace("Relic Unlocked: ", ""));
      }
    });
    setUnlockedRelics(relics);
  }, []);

  // --- AUDIO MUTING TOGGLE ---
  const handleToggleMute = useCallback(() => {
    playClickSound();
    const newMute = !muted;
    setMuted(newMute);
    setMuteState(newMute);
    saveGameStats({ audioMuted: newMute });
    postDialogue("SYSTEM", `Audio system ${newMute ? "Deactivated" : "Activated"}.`);
  }, [muted, postDialogue]);

  // --- KEYBOARD / CONTROLS BINDINGS ---
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (!isGameStartedRef.current) return;

      const key = e.key.toLowerCase();
      keysPressedRef.current[key] = true;

      // Prevent scroll with Arrow keys & Space
      if (["arrowup", "arrowdown", "arrowleft", "arrowright", " "].includes(e.key)) {
        e.preventDefault();
      }

      // Trigger Keyboard Mode
      if (["w", "a", "s", "d", "arrowup", "arrowdown", "arrowleft", "arrowright"].includes(key)) {
        usingKeyboardRef.current = true;
      }

      // 1-6 keys for formations
      if (["1", "2", "3", "4", "5", "6"].includes(key)) {
        playClickSound();
        const formationsList = Object.values(FormationType);
        const index = parseInt(key) - 1;
        const nextForm = formationsList[index];
        activeFormationRef.current = nextForm;
        setHudFormation(nextForm);
        postDialogue("COMPANION", `Recalibrated Swarm formation to: ${FORMATIONS[nextForm].name}.`);
      }

      // Pause toggle (P or Escape)
      if (key === "p" || key === "escape") {
        e.preventDefault();
        togglePause();
      }

      // Restart (R)
      if (key === "r" && isGameOverRef.current) {
        e.preventDefault();
        restartGame();
      }

      // Mute toggle (M)
      if (key === "m") {
        e.preventDefault();
        handleToggleMute();
      }
    };

    const handleKeyUp = (e: KeyboardEvent) => {
      const key = e.key.toLowerCase();
      keysPressedRef.current[key] = false;
    };

    const handleBlur = () => {
      keysPressedRef.current = {};
    };

    window.addEventListener("keydown", handleKeyDown);
    window.addEventListener("keyup", handleKeyUp);
    window.addEventListener("blur", handleBlur);
    document.addEventListener("visibilitychange", handleBlur);

    return () => {
      window.removeEventListener("keydown", handleKeyDown);
      window.removeEventListener("keyup", handleKeyUp);
      window.removeEventListener("blur", handleBlur);
      document.removeEventListener("visibilitychange", handleBlur);
    };
  }, [handleToggleMute, postDialogue]);

  // --- TOGGLE PAUSE ---
  const togglePause = () => {
    if (isGameOverRef.current) return;
    const nextPaused = !isPausedRef.current;
    isPausedRef.current = nextPaused;
    setIsPaused(nextPaused);
    if (nextPaused) {
      playPauseSound();
      postDialogue("COMPANION", "Swarm synchronization Suspended. Game Paused.");
    } else {
      playUnpauseSound();
      postDialogue("COMPANION", "Swarm synchronization Restored. Game Resumed.");
      lastTimeRef.current = performance.now(); // reset time to avoid huge deltaTime jump
    }
  };

  // --- ENTITY CREATORS & FX SHOWER ---
  const spawnExplosion = (x: number, y: number, color: string, count = 10) => {
    for (let i = 0; i < count; i++) {
      if (particlesRef.current.length >= MAX_PARTICLES) {
        particlesRef.current.shift();
      }
      const angle = Math.random() * Math.PI * 2;
      const speed = 1 + Math.random() * 4;
      particlesRef.current.push({
        id: Math.random().toString(),
        x,
        y,
        vx: Math.cos(angle) * speed,
        vy: Math.sin(angle) * speed,
        color,
        size: 2 + Math.random() * 3,
        life: 0,
        maxLife: 30 + Math.random() * 30
      });
    }
  };

  const spawnBiomeWarpFX = () => {
    const colors = ["#f59e0b", "#06b6d4", "#8b5cf6", "#10b981", "#ef4444", "#ec4899"];
    for (let i = 0; i < 40; i++) {
      if (particlesRef.current.length >= MAX_PARTICLES) particlesRef.current.shift();
      particlesRef.current.push({
        id: Math.random().toString(),
        x: Math.random() * CANVAS_WIDTH,
        y: Math.random() * CANVAS_HEIGHT,
        vx: (Math.random() - 0.5) * 8,
        vy: (Math.random() - 0.5) * 8,
        color: colors[Math.floor(Math.random() * colors.length)],
        size: 2 + Math.random() * 4,
        life: 0,
        maxLife: 40 + Math.random() * 40
      });
    }
  };

  const spawnResource = (type: "NANO" | "ENERGY" | "ORBI") => {
    if (resourcesRef.current.length >= MAX_RESOURCES) return;

    const x = 30 + Math.random() * (CANVAS_WIDTH - 60);
    const y = 30 + Math.random() * (CANVAS_HEIGHT - 60);
    
    let color = "#f59e0b"; // amber
    let amount = 1;
    let orbiType: OrbiType | undefined;

    if (type === "ENERGY") {
      color = "#06b6d4"; // cyan
    } else if (type === "ORBI") {
      color = "#ec4899"; // pink
      amount = 1;
      const typesList = [OrbiType.AURA, OrbiType.PULSE, OrbiType.SHIELD, OrbiType.BEAM, OrbiType.ECHO];
      orbiType = typesList[Math.floor(Math.random() * typesList.length)];
    }

    resourcesRef.current.push({
      id: Math.random().toString(),
      type,
      x,
      y,
      size: type === "ORBI" ? 10 : 6,
      color,
      amount,
      orbiType
    });
  };

  // --- BIOME SHIFTER ---
  const shiftToBiome = (biome: BiomeType) => {
    activeBiomeRef.current = biome;
    setHudBiome(biome);
    playBiomeChangeSound();
    spawnBiomeWarpFX();
    
    const config = BIOMES[biome];
    postDialogue("SYSTEM", `Warp completed: Arrived at [${config.name}].`);
    postDialogue("COMPANION", config.description);
  };

  // --- START THE GAME ---
  const startGame = () => {
    initAudio();
    isGameStartedRef.current = true;
    setIsGameStarted(true);
    
    scoreRef.current = 0;
    nanoCreditsRef.current = 0;
    purifiedEnergyRef.current = 0;
    
    playerRef.current.shield = PLAYER_MAX_SHIELD;
    playerRef.current.x = CANVAS_WIDTH / 2;
    playerRef.current.y = CANVAS_HEIGHT / 2;
    playerRef.current.invincibilityTimer = 0;
    
    // Check if player unlocked the Starting Orbi upgrade (Mitosis Relic)
    const startingOrbisCount = unlockedRelics.includes("Mitosis Relic") ? 3 : 1;
    
    swarmRef.current = [];
    // Add starting Foton leader
    swarmRef.current.push({
      id: "leader-foton",
      type: OrbiType.FOTON,
      name: ORBI_TEMPLATES[OrbiType.FOTON].name,
      x: playerRef.current.x,
      y: playerRef.current.y,
      targetX: playerRef.current.x,
      targetY: playerRef.current.y,
      color: ORBI_TEMPLATES[OrbiType.FOTON].color,
      size: ORBI_TEMPLATES[OrbiType.FOTON].size,
      shootCooldown: 0,
      pulseTimer: 0
    });

    // Add extra starter orbis if upgrade unlocked
    if (startingOrbisCount > 1) {
      for (let i = 1; i < startingOrbisCount; i++) {
        const types = [OrbiType.AURA, OrbiType.PULSE];
        const t = types[i - 1] || OrbiType.AURA;
        swarmRef.current.push({
          id: `starter-follower-${i}`,
          type: t,
          name: ORBI_TEMPLATES[t].name,
          x: playerRef.current.x + (Math.random() - 0.5) * 40,
          y: playerRef.current.y + (Math.random() - 0.5) * 40,
          targetX: playerRef.current.x,
          targetY: playerRef.current.y,
          color: ORBI_TEMPLATES[t].color,
          size: ORBI_TEMPLATES[t].size,
          shootCooldown: 0,
          pulseTimer: 0
        });
      }
    }

    enemiesRef.current = [];
    projectilesRef.current = [];
    resourcesRef.current = [];
    particlesRef.current = [];
    
    activeBiomeRef.current = BiomeType.SOLAR_PLAINS;
    setHudBiome(BiomeType.SOLAR_PLAINS);
    activeFormationRef.current = FormationType.CIRCLE;
    setHudFormation(FormationType.CIRCLE);
    
    isPausedRef.current = false;
    setIsPaused(false);
    isGameOverRef.current = false;
    setIsGameOver(false);

    // Initial resources
    for (let i = 0; i < 10; i++) spawnResource("NANO");
    for (let i = 0; i < 5; i++) spawnResource("ENERGY");
    spawnResource("ORBI");

    lastTimeRef.current = performance.now();
    postDialogue("SYSTEM", "Grid connection established successfully.");
    postDialogue("COMPANION", "Foton active! Protect our enigm-core and absorb glowing crystals.");
  };

  // --- RESTART GAME ---
  const restartGame = () => {
    playClickSound();
    startGame();
  };

  // --- END GAME (DEFEAT) ---
  const handleGameOver = () => {
    if (isGameOverRef.current) return;
    isGameOverRef.current = true;
    setIsGameOver(true);
    playGameOverSound();

    // Calculate final stats and persistent storage updates
    const currentHighScore = Math.max(hudHighScore, scoreRef.current);
    setHudHighScore(currentHighScore);

    const loaded = loadGameStats();
    const totalRuns = loaded.totalRuns + 1;
    const totalNano = loaded.totalNanoCredits + nanoCreditsRef.current;
    const totalEnergy = loaded.totalPurifiedEnergy + purifiedEnergyRef.current;
    const bestSwarm = Math.max(loaded.bestSwarmSize, swarmRef.current.length);

    // Create a new memory sometimes
    const newMemories = [...loaded.gameMemories];
    if (scoreRef.current > 2000 && !newMemories.includes("Quantum Leap: Surpassed 2000 Units of Grid Energy.")) {
      newMemories.push("Quantum Leap: Surpassed 2000 Units of Grid Energy.");
    }
    if (swarmRef.current.length >= 10 && !newMemories.includes("Megaswarm: Orchestrated 10+ Orbis simultanously.")) {
      newMemories.push("Megaswarm: Orchestrated 10+ Orbis simultanously.");
    }

    saveGameStats({
      highScore: currentHighScore,
      totalNanoCredits: totalNano,
      totalPurifiedEnergy: totalEnergy,
      bestSwarmSize: bestSwarm,
      totalRuns: totalRuns,
      gameMemories: newMemories
    });

    // Sync React Upgrade state
    setStats({
      ...loaded,
      highScore: currentHighScore,
      totalNanoCredits: totalNano,
      totalPurifiedEnergy: totalEnergy,
      bestSwarmSize: bestSwarm,
      totalRuns: totalRuns,
      gameMemories: newMemories
    });

    postDialogue("SYSTEM", "CRITICAL ERROR: Connection to Orbi Grid dropped.");
    postDialogue("COMPANION", "Foton! Orbi core shattered... Press [R] to reboot our grid connection.");
  };

  // --- UPGRADE RELIC HANDLER ---
  const handleBuyRelic = (relicName: string, cost: number) => {
    const loaded = loadGameStats();
    if (loaded.totalNanoCredits >= cost && !unlockedRelics.includes(relicName)) {
      playRecruitSound();
      const updatedCredits = loaded.totalNanoCredits - cost;
      const updatedMemories = [...loaded.gameMemories, `Relic Unlocked: ${relicName}`];
      
      saveGameStats({
        totalNanoCredits: updatedCredits,
        gameMemories: updatedMemories
      });

      setStats(prev => prev ? {
        ...prev,
        totalNanoCredits: updatedCredits,
        gameMemories: updatedMemories
      } : null);

      setUnlockedRelics(prev => [...prev, relicName]);
      postDialogue("COMPANION", `Unlocked relic: ${relicName}. Swarm stats hardened!`);
    } else {
      playDamageSound();
    }
  };

  // --- MOBILE DIRECTION PAD INTERACTION ---
  const handleDPadPress = (dx: number, dy: number) => {
    if (!isGameStartedRef.current || isPausedRef.current || isGameOverRef.current) return;
    usingKeyboardRef.current = true;
    playerRef.current.x += dx * 20;
    playerRef.current.y += dy * 20;
    
    // Keep inside bounds
    playerRef.current.x = Math.max(15, Math.min(CANVAS_WIDTH - 15, playerRef.current.x));
    playerRef.current.y = Math.max(15, Math.min(CANVAS_HEIGHT - 15, playerRef.current.y));
    
    pointerRef.current = { x: playerRef.current.x, y: playerRef.current.y };
  };

  // --- CORE SIMULATION & MAIN GAME LOOP (unified requestAnimationFrame) ---
  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    const mainGameLoop = (timestamp: number) => {
      loopIdRef.current = requestAnimationFrame(mainGameLoop);

      if (!lastTimeRef.current) {
        lastTimeRef.current = timestamp;
      }
      // Limit delta time to prevent giant frame leaps during tab suspension
      let delta = timestamp - lastTimeRef.current;
      if (delta > 100) delta = 16.66; // fallback roughly 60fps
      lastTimeRef.current = timestamp;

      // Clear Canvas
      const currentBiome = activeBiomeRef.current;
      const biomeConfig = BIOMES[currentBiome];
      ctx.fillStyle = biomeConfig.bgColor;
      ctx.fillRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);

      // Scroll background grid lines
      drawGrid(ctx, biomeConfig.gridColor);

      if (!isGameStartedRef.current) {
        // Just draw nice star background on initial screen
        updateAndDrawParticles(ctx, delta, false);
        return;
      }

      const isGamePaused = isPausedRef.current;
      const isGameEnded = isGameOverRef.current;

      // 1. UPDATE GAME STATS (Throttled synchronization to React State)
      frameCountRef.current++;
      if (frameCountRef.current % 10 === 0) {
        setHudShield(Math.round(playerRef.current.shield));
        setHudScore(scoreRef.current);
        setHudPopulation(swarmRef.current.length);
        setHudNano(nanoCreditsRef.current);
        setHudEnergy(purifiedEnergyRef.current);
      }

      // Propose random dialogues from LUX-8 companion every 15 seconds
      if (timestamp - lastDialogueTimeRef.current > 15000 && !isGamePaused && !isGameEnded) {
        lastDialogueTimeRef.current = timestamp;
        const randomQuote = COMPANION_QUOTES[Math.floor(Math.random() * COMPANION_QUOTES.length)];
        postDialogue("COMPANION", randomQuote);
      }

      if (!isGamePaused && !isGameEnded) {
        // --- HIGH FREQUENCY PHYSICS UPDATES (ONLY when active) ---

        // A. Apply Player Invincibility Frame Countdowns
        if (playerRef.current.invincibilityTimer > 0) {
          playerRef.current.invincibilityTimer -= delta;
        }

        // B. Dynamic Player Movement (Keys OR pointer follow)
        let moveX = 0;
        let moveY = 0;

        if (keysPressedRef.current["w"] || keysPressedRef.current["arrowup"]) moveY -= 1;
        if (keysPressedRef.current["s"] || keysPressedRef.current["arrowdown"]) moveY += 1;
        if (keysPressedRef.current["a"] || keysPressedRef.current["arrowleft"]) moveX -= 1;
        if (keysPressedRef.current["d"] || keysPressedRef.current["arrowright"]) moveX += 1;

        // Apply Speed Upgrades if Relic purchased
        let speedMultiplier = 1.0;
        if (unlockedRelics.includes("Quantum Core")) speedMultiplier = 1.15;

        // Apply Biome Speeds
        const biomeSpeedMod = biomeConfig.enemySpeedMultiplier; // Water slows, Swamp speeds

        if (moveX !== 0 || moveY !== 0) {
          usingKeyboardRef.current = true;
          // Normalize movement vector
          const len = Math.sqrt(moveX * moveX + moveY * moveY);
          const finalSpeed = playerRef.current.speed * speedMultiplier * (biomeSpeedMod < 1 ? 0.75 : 1);
          playerRef.current.x += (moveX / len) * finalSpeed;
          playerRef.current.y += (moveY / len) * finalSpeed;

          // Keep within bounds
          playerRef.current.x = Math.max(15, Math.min(CANVAS_WIDTH - 15, playerRef.current.x));
          playerRef.current.y = Math.max(15, Math.min(CANVAS_HEIGHT - 15, playerRef.current.y));

          // Feed keyboard location to grid scroll offset
          gridOffsetRef.current.x -= (moveX / len) * finalSpeed * 0.5;
          gridOffsetRef.current.y -= (moveY / len) * finalSpeed * 0.5;
        } else if (!usingKeyboardRef.current) {
          // Smooth Mouse/Touch tracking
          const dx = pointerRef.current.x - playerRef.current.x;
          const dy = pointerRef.current.y - playerRef.current.y;
          const dist = Math.sqrt(dx * dx + dy * dy);
          if (dist > 3) {
            const dragSpeed = 0.08 * (biomeSpeedMod < 1 ? 0.75 : 1);
            playerRef.current.x += dx * dragSpeed;
            playerRef.current.y += dy * dragSpeed;
            
            gridOffsetRef.current.x -= dx * dragSpeed * 0.5;
            gridOffsetRef.current.y -= dy * dragSpeed * 0.5;
          }
        }

        // C. Regenerate Player Shield
        if (playerRef.current.shield < PLAYER_MAX_SHIELD) {
          // Increase regen rate if Deep Water biome active OR deflector relic unlocked
          let regen = PLAYER_SHIELD_REGEN;
          if (currentBiome === BiomeType.DEEP_WATER) regen *= 2;
          if (unlockedRelics.includes("Hydrogen Deflector")) regen *= 1.5;
          playerRef.current.shield = Math.min(PLAYER_MAX_SHIELD, playerRef.current.shield + regen);
        }

        // D. Calculate Follower Trailing Targets based on active Formation
        updateSwarmFormations(timestamp);

        // E. Automatic Follower Weapon Systems
        handleSwarmWeapons();

        // F. Simulate enemies
        handleEnemies(delta);

        // G. Projectile physics and bounds
        handleProjectiles(delta);

        // H. Resources collection and attraction
        handleResources();

        // I. Periodically spawn enemies & resources
        if (frameCountRef.current % 180 === 0) {
          // spawn resources
          const rng = Math.random();
          if (rng < 0.6) spawnResource("NANO");
          else if (rng < 0.9) spawnResource("ENERGY");
          else spawnResource("ORBI");
        }

        if (Math.random() < biomeConfig.enemySpawnRate && enemiesRef.current.length < MAX_ENEMIES) {
          spawnRandomEnemy();
        }
      }

      // 2. RENDERING GAME OBJECTS (Continuous visual processing)
      // A. Particles
      updateAndDrawParticles(ctx, delta, !isGamePaused && !isGameEnded);

      // B. Draw Resources
      drawResources(ctx);

      // C. Draw Player Projectiles
      drawProjectiles(ctx);

      // D. Draw Enemies
      drawEnemies(ctx);

      // E. Draw Swarm Followers & Leader
      drawSwarm(ctx, timestamp);

      // F. Invincibility flash overlay
      if (playerRef.current.invincibilityTimer > 0 && Math.floor(timestamp / 100) % 2 === 0) {
        ctx.fillStyle = "rgba(239, 68, 68, 0.15)"; // Soft red flashing indicator
        ctx.fillRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);
      }
    };

    loopIdRef.current = requestAnimationFrame(mainGameLoop);

    return () => {
      if (loopIdRef.current) {
        cancelAnimationFrame(loopIdRef.current);
      }
    };
  }, [unlockedRelics, postDialogue]);

  // --- DRAW HORIZON GRID LINES ---
  const drawGrid = (ctx: CanvasRenderingContext2D, gridColor: string) => {
    ctx.strokeStyle = gridColor;
    ctx.lineWidth = 1;

    const spacing = 40;
    const scrollX = gridOffsetRef.current.x % spacing;
    const scrollY = gridOffsetRef.current.y % spacing;

    // Draw vertical grid lines
    for (let x = scrollX; x < CANVAS_WIDTH; x += spacing) {
      ctx.beginPath();
      ctx.moveTo(x, 0);
      ctx.lineTo(x, CANVAS_HEIGHT);
      ctx.stroke();
    }

    // Draw horizontal grid lines
    for (let y = scrollY; y < CANVAS_HEIGHT; y += spacing) {
      ctx.beginPath();
      ctx.moveTo(0, y);
      ctx.lineTo(CANVAS_WIDTH, y);
      ctx.stroke();
    }
  };

  // --- SWARM FORMATION POSITIONING CALCULATOR ---
  const updateSwarmFormations = (time: number) => {
    const leader = playerRef.current;
    const members = swarmRef.current;
    const formType = activeFormationRef.current;
    const spacing = FORMATIONS[formType].spacing;
    const N = members.length;

    if (N === 0) return;

    // Position Foton exactly at Player
    members[0].x = leader.x;
    members[0].y = leader.y;

    if (N === 1) return;

    // Calculate movement vector angle of leader for rotating formations
    let angle = 0;
    if (usingKeyboardRef.current) {
      // Calculate angle from pressed keys
      let mx = 0; let my = 0;
      if (keysPressedRef.current["w"] || keysPressedRef.current["arrowup"]) my -= 1;
      if (keysPressedRef.current["s"] || keysPressedRef.current["arrowdown"]) my += 1;
      if (keysPressedRef.current["a"] || keysPressedRef.current["arrowleft"]) mx -= 1;
      if (keysPressedRef.current["d"] || keysPressedRef.current["arrowright"]) mx += 1;
      if (mx !== 0 || my !== 0) {
        angle = Math.atan2(my, mx);
      }
    } else {
      // Calculate angle from mouse vector
      angle = Math.atan2(pointerRef.current.y - leader.y, pointerRef.current.x - leader.x);
    }

    for (let i = 1; i < N; i++) {
      let targetX = leader.x;
      let targetY = leader.y;

      switch (formType) {
        case FormationType.LINE: {
          // Linear sweep offset, spaced on left/right vectors relative to movement direction
          const offsetIndex = Math.ceil(i / 2) * (i % 2 === 0 ? 1 : -1);
          const localDx = offsetIndex * spacing;
          const localDy = 25; // slightly behind Foton

          // Rotate local offset vectors
          const rotX = localDx * Math.cos(angle - Math.PI/2) - localDy * Math.sin(angle - Math.PI/2);
          const rotY = localDx * Math.sin(angle - Math.PI/2) + localDy * Math.cos(angle - Math.PI/2);

          targetX = leader.x + rotX;
          targetY = leader.y + rotY;
          break;
        }
        case FormationType.CIRCLE: {
          // Spinning ring around the leader
          const speed = 0.0015;
          const offsetAngle = (i * 2 * Math.PI) / (N - 1) + time * speed;
          targetX = leader.x + Math.cos(offsetAngle) * spacing;
          targetY = leader.y + Math.sin(offsetAngle) * spacing;
          break;
        }
        case FormationType.DELTA: {
          // Sharp triangular delta spearhead trailing behind leader
          const row = Math.floor((Math.sqrt(8 * i + 1) - 1) / 2);
          const col = i - (row * (row + 1)) / 2;
          const rowSpacing = spacing;
          const colSpacing = spacing;

          const localDx = (col - row / 2) * colSpacing;
          const localDy = -row * rowSpacing - 10; // trail behind

          // Rotate vectors
          const rotX = localDx * Math.cos(angle + Math.PI) - localDy * Math.sin(angle + Math.PI);
          const rotY = localDx * Math.sin(angle + Math.PI) + localDy * Math.cos(angle + Math.PI);

          targetX = leader.x + rotX;
          targetY = leader.y + rotY;
          break;
        }
        case FormationType.SHIELD: {
          // Tight halo protective shield
          const offsetAngle = (i * 2 * Math.PI) / (N - 1);
          targetX = leader.x + Math.cos(offsetAngle) * spacing;
          targetY = leader.y + Math.sin(offsetAngle) * spacing;
          break;
        }
        case FormationType.V_SHAPE: {
          // Classic fighter jet V-Shape flanking
          const side = i % 2 === 0 ? 1 : -1;
          const indexOffset = Math.ceil(i / 2);
          const localDx = side * indexOffset * spacing * 0.7;
          const localDy = -indexOffset * spacing;

          // Rotate vectors
          const rotX = localDx * Math.cos(angle + Math.PI) - localDy * Math.sin(angle + Math.PI);
          const rotY = localDx * Math.sin(angle + Math.PI) + localDy * Math.cos(angle + Math.PI);

          targetX = leader.x + rotX;
          targetY = leader.y + rotY;
          break;
        }
        case FormationType.SCATTERED: {
          // Loosely floating organic offset using sine waves
          const phase = i * 137.5 * (Math.PI / 180);
          const offsetDist = spacing + Math.sin(time * 0.002 + i) * 12;
          targetX = leader.x + Math.cos(phase + Math.sin(time * 0.001 + i) * 0.5) * offsetDist;
          targetY = leader.y + Math.sin(phase + Math.cos(time * 0.001 + i) * 0.5) * offsetDist;
          break;
        }
      }

      // Interpolate position smoothly to prevent sudden visual jumping
      members[i].x += (targetX - members[i].x) * 0.08;
      members[i].y += (targetY - members[i].y) * 0.08;
    }
  };

  // --- SWARM AUTOSHOOT LOGIC ---
  const handleSwarmWeapons = () => {
    // Collect non-dead enemies
    const activeEnemies = enemiesRef.current.filter(e => !e.isDead);
    if (activeEnemies.length === 0) return;

    // Relic speed modifier
    let cdModifier = 1.0;
    if (unlockedRelics.includes("Quantum Core")) cdModifier = 0.75; // 25% faster weapon cycles

    swarmRef.current.forEach((member) => {
      // Weapon cooldown cycle
      if (member.shootCooldown > 0) {
        member.shootCooldown--;
        return;
      }

      // Find closest enemy
      let closestEnemy: Enemy | null = null;
      let minDist = 300; // max shoot range

      activeEnemies.forEach((enemy) => {
        const dx = enemy.x - member.x;
        const dy = enemy.y - member.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < minDist) {
          minDist = dist;
          closestEnemy = enemy;
        }
      });

      if (closestEnemy) {
        // Aim and fire
        const enemy: Enemy = closestEnemy;
        const dx = enemy.x - member.x;
        const dy = enemy.y - member.y;
        const dist = Math.sqrt(dx * dx + dy * dy);

        // Set speed of projectile
        const projSpeed = 8;
        const vx = (dx / dist) * projSpeed;
        const vy = (dy / dist) * projSpeed;

        if (projectilesRef.current.length < MAX_PROJECTILES) {
          projectilesRef.current.push({
            id: Math.random().toString(),
            x: member.x,
            y: member.y,
            vx,
            vy,
            damage: member.type === OrbiType.FOTON ? 15 : 10,
            color: member.color,
            size: member.type === OrbiType.FOTON ? 5 : 3.5,
            fromPlayer: true,
            life: 0,
            maxLife: 60
          });
          playShootSound();
        }

        // Set base cooldown (Foton fires faster, followers slower)
        const baseCd = member.type === OrbiType.FOTON ? 25 : 45;
        member.shootCooldown = Math.round(baseCd * cdModifier);
      }
    });
  };

  // --- SPAWN RANDOM ENEMIES ---
  const spawnRandomEnemy = () => {
    const side = Math.floor(Math.random() * 4);
    let x = 0; let y = 0;
    
    // Pick borders
    if (side === 0) { // Top
      x = Math.random() * CANVAS_WIDTH;
      y = -10;
    } else if (side === 1) { // Right
      x = CANVAS_WIDTH + 10;
      y = Math.random() * CANVAS_HEIGHT;
    } else if (side === 2) { // Bottom
      x = Math.random() * CANVAS_WIDTH;
      y = CANVAS_HEIGHT + 10;
    } else { // Left
      x = -10;
      y = Math.random() * CANVAS_HEIGHT;
    }

    const types = [EnemyType.CRAWLER, EnemyType.PARASITE, EnemyType.DRONE];
    const pickedType = types[Math.floor(Math.random() * types.length)];
    
    let hp = 20;
    let speed = 1.5;
    let size = 10;
    let color = "#ef4444"; // red Crawler

    if (pickedType === EnemyType.PARASITE) {
      hp = 10;
      speed = 2.4;
      size = 7;
      color = "#a855f7"; // purple Parasite
    } else if (pickedType === EnemyType.DRONE) {
      hp = 30;
      speed = 1.0;
      size = 12;
      color = "#3b82f6"; // blue Drone
    }

    // Multiply speeds and health slightly over score
    const scaling = 1 + (scoreRef.current / 4000);

    enemiesRef.current.push({
      id: Math.random().toString(),
      type: pickedType,
      x,
      y,
      vx: 0,
      vy: 0,
      health: hp * scaling,
      maxHealth: hp * scaling,
      speed: speed * (BIOMES[activeBiomeRef.current].enemySpeedMultiplier),
      size,
      color,
      shootCooldown: Math.round(90 + Math.random() * 90),
      contactDamageCooldown: 0,
      isDead: false
    });
  };

  // --- ENEMIES TICK & AI ---
  const handleEnemies = (delta: number) => {
    const player = playerRef.current;
    
    enemiesRef.current.forEach((enemy) => {
      if (enemy.isDead) return;

      // Update contact cooldowns
      if (enemy.contactDamageCooldown > 0) {
        enemy.contactDamageCooldown -= delta;
      }

      const dx = player.x - enemy.x;
      const dy = player.y - enemy.y;
      const dist = Math.sqrt(dx * dx + dy * dy);

      switch (enemy.type) {
        case EnemyType.CRAWLER: {
          // Direct relentless pursuit
          enemy.vx = (dx / dist) * enemy.speed;
          enemy.vy = (dy / dist) * enemy.speed;
          break;
        }
        case EnemyType.PARASITE: {
          // Jumpy behavior
          const jumpPhase = Math.sin(frameCountRef.current * 0.08) * 0.5 + 1.0;
          enemy.vx = (dx / dist) * enemy.speed * jumpPhase;
          enemy.vy = (dy / dist) * enemy.speed * jumpPhase;
          break;
        }
        case EnemyType.DRONE: {
          // Keep distant and orbit
          if (dist > 220) {
            enemy.vx = (dx / dist) * enemy.speed;
            enemy.vy = (dy / dist) * enemy.speed;
          } else if (dist < 150) {
            enemy.vx = -(dx / dist) * enemy.speed;
            enemy.vy = -(dy / dist) * enemy.speed;
          } else {
            // Orbit circle around Foton
            enemy.vx = (-dy / dist) * enemy.speed;
            enemy.vy = (dx / dist) * enemy.speed;
          }

          // Shoot drone projectile occasionally
          if (enemy.shootCooldown > 0) {
            enemy.shootCooldown--;
          } else {
            enemy.shootCooldown = 150 + Math.random() * 100;
            if (projectilesRef.current.length < MAX_PROJECTILES) {
              const projSpeed = 4.5;
              projectilesRef.current.push({
                id: Math.random().toString(),
                x: enemy.x,
                y: enemy.y,
                vx: (dx / dist) * projSpeed,
                vy: (dy / dist) * projSpeed,
                damage: 12,
                color: "#f43f5e", // dangerous rose laser
                size: 4,
                fromPlayer: false,
                life: 0,
                maxLife: 180
              });
              playEnemyShootSound();
            }
          }
          break;
        }
      }

      // Apply positions
      enemy.x += enemy.vx;
      enemy.y += enemy.vy;

      // Contact Damage check (Foton invincibility frame respected)
      if (dist < (player.radius + enemy.size)) {
        if (player.invincibilityTimer <= 0) {
          let damage = 20;
          if (enemy.type === EnemyType.PARASITE) damage = 12; // lower
          player.shield = Math.max(0, player.shield - damage);
          player.invincibilityTimer = DAMAGE_INVINCIBILITY_DURATION;
          playDamageSound();
          spawnExplosion(player.x, player.y, "#ef4444", 8);

          if (player.shield <= 0) {
            handleGameOver();
          }
        }
      }
    });

    // Clean up dead enemies or out of bounds
    enemiesRef.current = enemiesRef.current.filter((e) => {
      if (e.isDead) return false;
      // If incredibly far offbounds, remove
      if (e.x < -100 || e.x > CANVAS_WIDTH + 100 || e.y < -100 || e.y > CANVAS_HEIGHT + 100) {
        return false;
      }
      return true;
    });
  };

  // --- PROJECTILE SYSTEMS PHYSICS & SAFE COLLISIONS ---
  const handleProjectiles = (_delta: number) => {
    projectilesRef.current.forEach((proj) => {
      proj.x += proj.vx;
      proj.y += proj.vy;
      proj.life += 1;

      // Collisions checks
      if (proj.fromPlayer) {
        // Player weapon hitting enemies
        enemiesRef.current.forEach((enemy) => {
          if (enemy.isDead) return;
          const dx = enemy.x - proj.x;
          const dy = enemy.y - proj.y;
          const dist = Math.sqrt(dx * dx + dy * dy);

          if (dist < (enemy.size + proj.size)) {
            // SAFE HIT COLLISION: Mark projectile as dead instantly to prevent hit repetitions
            proj.life = proj.maxLife + 1;
            enemy.health -= proj.damage;
            
            // Spawn hit sparks
            spawnExplosion(proj.x, proj.y, proj.color, 3);

            if (enemy.health <= 0) {
              enemy.isDead = true;
              playEnemyExplodeSound();
              spawnExplosion(enemy.x, enemy.y, enemy.color, 12);
              
              // Increment score & award credits
              const crystalMultiplier = BIOMES[activeBiomeRef.current].crystalValueMultiplier;
              const pointsGained = Math.round((enemy.type === EnemyType.DRONE ? 150 : 80) * crystalMultiplier);
              scoreRef.current += pointsGained;

              // Progress towards next biome shifts
              biomeProgressRef.current += pointsGained;
              if (biomeProgressRef.current >= 2000) {
                biomeProgressRef.current = 0;
                // Shift to random new biome
                const types = Object.values(BiomeType);
                const filtered = types.filter(t => t !== activeBiomeRef.current);
                const randomNext = filtered[Math.floor(Math.random() * filtered.length)];
                shiftToBiome(randomNext);
              }
            }
          }
        });
      } else {
        // Enemy weapon hitting Foton player
        const player = playerRef.current;
        const dx = player.x - proj.x;
        const dy = player.y - proj.y;
        const dist = Math.sqrt(dx * dx + dy * dy);

        if (dist < (player.radius + proj.size)) {
          proj.life = proj.maxLife + 1; // kill proj
          if (player.invincibilityTimer <= 0) {
            player.shield = Math.max(0, player.shield - proj.damage);
            player.invincibilityTimer = DAMAGE_INVINCIBILITY_DURATION;
            playDamageSound();
            spawnExplosion(player.x, player.y, "#ef4444", 8);

            if (player.shield <= 0) {
              handleGameOver();
            }
          }
        }
      }
    });

    // Remove dead projectiles
    projectilesRef.current = projectilesRef.current.filter((p) => {
      if (p.life > p.maxLife) return false;
      if (p.x < 0 || p.x > CANVAS_WIDTH || p.y < 0 || p.y > CANVAS_HEIGHT) return false;
      return true;
    });
  };

  // --- RESOURCES MANAGEMENT AND ATTRACTION (MAGNET EFFECT) ---
  const handleResources = () => {
    const player = playerRef.current;
    
    // Attraction radius (Magnetic Relic increases it)
    let magnetRadius = 60;
    if (unlockedRelics.includes("Fusion Resonator")) magnetRadius = 110;

    resourcesRef.current.forEach((res) => {
      const dx = player.x - res.x;
      const dy = player.y - res.y;
      const dist = Math.sqrt(dx * dx + dy * dy);

      // Magnetic pull to player
      if (dist < magnetRadius) {
        const pullPower = (magnetRadius - dist) / magnetRadius * 4;
        res.x += (dx / dist) * pullPower;
        res.y += (dy / dist) * pullPower;
      }

      // Collect items
      if (dist < (player.radius + res.size)) {
        res.size = 0; // mark collected

        if (res.type === "NANO") {
          let value = res.amount;
          // Solar biome doubles nano credits
          if (activeBiomeRef.current === BiomeType.SOLAR_PLAINS) value *= 2;
          // Nano catalyst relic upgrade doubles nano credits
          if (unlockedRelics.includes("Nano Catalyst")) value *= 2;
          
          nanoCreditsRef.current += value;
          scoreRef.current += 50;
          playCrystalSound();
          spawnExplosion(res.x, res.y, res.color, 5);
        } else if (res.type === "ENERGY") {
          let value = res.amount;
          if (unlockedRelics.includes("Nano Catalyst")) value *= 1.5;
          
          purifiedEnergyRef.current += Math.round(value);
          scoreRef.current += 150;
          playCrystalSound();
          spawnExplosion(res.x, res.y, res.color, 5);
        } else if (res.type === "ORBI") {
          // Recruit a new Orbi member
          if (swarmRef.current.length < MAX_SWARM_MEMBERS) {
            const newType = res.orbiType || OrbiType.AURA;
            swarmRef.current.push({
              id: Math.random().toString(),
              type: newType,
              name: ORBI_TEMPLATES[newType].name,
              x: res.x,
              y: res.y,
              targetX: player.x,
              targetY: player.y,
              color: ORBI_TEMPLATES[newType].color,
              size: ORBI_TEMPLATES[newType].size,
              shootCooldown: 30,
              pulseTimer: 0
            });
            playRecruitSound();
            spawnExplosion(res.x, res.y, "#ec4899", 15); // Big pink sparks
            postDialogue("COMPANION", `Recruited wild [${ORBI_TEMPLATES[newType].name}]! Firepower increased.`);
          }
        }
      }
    });

    // Remove collected resources
    resourcesRef.current = resourcesRef.current.filter(r => r.size > 0);
  };

  // --- PARTICLE ANIMATION AND DRAWING ---
  const updateAndDrawParticles = (ctx: CanvasRenderingContext2D, delta: number, simulate: boolean) => {
    particlesRef.current.forEach((p) => {
      if (simulate) {
        p.x += p.vx;
        p.y += p.vy;
        p.life += delta * 0.06; // age particles relative to time
      } else {
        // slow drift when paused
        p.x += p.vx * 0.05;
        p.y += p.vy * 0.05;
      }

      const alpha = Math.max(0, 1 - (p.life / p.maxLife));
      ctx.fillStyle = p.color;
      ctx.globalAlpha = alpha;
      ctx.beginPath();
      ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
      ctx.fill();
    });

    ctx.globalAlpha = 1.0; // reset transparency

    // Filter dead particles
    particlesRef.current = particlesRef.current.filter(p => p.life < p.maxLife);
  };

  // --- CANVAS DRAWING SYSTEMS ---
  const drawResources = (ctx: CanvasRenderingContext2D) => {
    resourcesRef.current.forEach((res) => {
      ctx.save();
      
      // Draw outer glowing ring
      ctx.shadowBlur = 10;
      ctx.shadowColor = res.color;
      
      if (res.type === "ORBI") {
        // Draw spinning hexagon star for Orbi crystals
        const angle = (frameCountRef.current * 0.03);
        ctx.strokeStyle = res.color;
        ctx.fillStyle = "rgba(236, 72, 153, 0.3)";
        ctx.lineWidth = 2;
        ctx.beginPath();
        for (let i = 0; i < 6; i++) {
          const a = angle + (i * Math.PI) / 3;
          const rx = res.x + Math.cos(a) * res.size;
          const ry = res.y + Math.sin(a) * res.size;
          if (i === 0) ctx.moveTo(rx, ry);
          else ctx.lineTo(rx, ry);
        }
        ctx.closePath();
        ctx.fill();
        ctx.stroke();

        // Inner core
        ctx.fillStyle = "#fff";
        ctx.beginPath();
        ctx.arc(res.x, res.y, 4, 0, Math.PI * 2);
        ctx.fill();
      } else {
        // Draw triangular or diamond crystals
        ctx.fillStyle = res.color;
        ctx.beginPath();
        ctx.moveTo(res.x, res.y - res.size);
        ctx.lineTo(res.x + res.size, res.y);
        ctx.lineTo(res.x, res.y + res.size);
        ctx.lineTo(res.x - res.size, res.y);
        ctx.closePath();
        ctx.fill();
      }

      ctx.restore();
    });
  };

      ctx.strokeStyle = proj.color;

          const lateralWobble = Math.sin(speedOffset + j) * 4;

          const sx = enemy.x - xOffset + Math.cos(angle + Math.PI/2) * lateralWobble;
          const sy = enemy.y - yOffset + Math.sin(angle + Math.PI/2) * lateralWobble;

          ctx.beginPath();
          ctx.arc(sx, sy, enemy.size * (1 - j * 0.2), 0, Math.PI * 2);
          ctx.fill();

        ctx.lineWidth = 1.5;

          const a = (i * Math.PI) / 2 + twitchAngle;

          ctx.moveTo(enemy.x, enemy.y);
          ctx.lineTo(enemy.x + Math.cos(a) * (enemy.size + 4), enemy.y + Math.sin(a) * (enemy.size + 4));

        }

        ctx.moveTo(enemy.x, enemy.y - enemy.size * 1.3);
        ctx.lineTo(enemy.x + enemy.size * 1.1, enemy.y);
        ctx.lineTo(enemy.x, enemy.y + enemy.size * 1.3);
        ctx.lineTo(enemy.x - enemy.size * 1.1, enemy.y);

        ctx.fill();

        ctx.beginPath();
        ctx.arc(enemy.x, enemy.y, 3, 0, Math.PI * 2);
        ctx.fill();
      }

      ctx.restore();
    });

      ctx.arc(lx, ly, eyeRadius, 0, Math.PI * 2);
      ctx.fill();

      // Pupil

      const ply = ly + Math.sin(angle) * (eyeRadius * 0.4);
      ctx.fillStyle = "#090d16";
      ctx.beginPath();
      ctx.arc(plx, ply, pupilRadius, 0, Math.PI * 2);
      ctx.fill();

      // Right Eye
      const rx = member.x + Math.cos(angle + 0.5) * lookDist;
      const ry = member.y + Math.sin(angle + 0.5) * lookDist;
      ctx.fillStyle = "#ffffff";
      ctx.beginPath();
      ctx.arc(rx, ry, eyeRadius, 0, Math.PI * 2);
      ctx.fill();

      // Pupil
      const prx = rx + Math.cos(angle) * (eyeRadius * 0.4);
      const pry = ry + Math.sin(angle) * (eyeRadius * 0.4);
      ctx.fillStyle = "#090d16";
      ctx.beginPath();
      ctx.arc(prx, pry, pupilRadius, 0, Math.PI * 2);
      ctx.fill();

        ctx.lineWidth = 1.5;
        ctx.setLineDash([4, 4]);
        ctx.beginPath();
        ctx.arc(member.x, member.y, member.size * 2, 0, Math.PI * 2);
        ctx.stroke();
      }

      ctx.restore();
    });

import { GameStats } from "./types";

const STORAGE_KEY = "orbi_energy_swarm_save_v1";

const DEFAULT_STATS: GameStats = {
  score: 0,
  highScore: 0,
  nanoCredits: 0,
  purifiedEnergy: 0,
  swarmSize: 1,
  bestSwarmSize: 1,
  totalNanoCredits: 0,
  totalPurifiedEnergy: 0,
  totalRuns: 0,
  gameMemories: [
    "Grid Connection Established.",
    "First Swarm Synthesis Activated."
  ],
  audioMuted: false
};

export const loadGameStats = (): GameStats => {
  try {
    const data = localStorage.getItem(STORAGE_KEY);
    if (!data) return { ...DEFAULT_STATS };
    const parsed = JSON.parse(data);
    return {
      ...DEFAULT_STATS,
      ...parsed,
      // Ensure arrays and primitive fields are present
      gameMemories: Array.isArray(parsed.gameMemories) ? parsed.gameMemories : DEFAULT_STATS.gameMemories
    };
  } catch (error) {
    console.error("Failed to load ORBI ENERGY SWARM local storage:", error);
    return { ...DEFAULT_STATS };
  }
};

export const saveGameStats = (stats: Partial<GameStats>): void => {
  try {
    const current = loadGameStats();
    const updated = { ...current, ...stats };

let audioCtx: AudioContext | null = null;
let isMuted = false;

export const initAudio = (): AudioContext | null => {
  if (isMuted) return null;
  if (!audioCtx) {
    try {
      const AudioCtxClass = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
      audioCtx = new AudioCtxClass();
    } catch (e) {
      console.error("Web Audio API not supported:", e);
    }
  }
  if (audioCtx && audioCtx.state === "suspended") {
    audioCtx.resume().catch((err) => console.warn("Failed to resume AudioContext:", err));
  }
  return audioCtx;
};

export const setMuteState = (muted: boolean) => {
  isMuted = muted;
  if (muted && audioCtx) {
    audioCtx.suspend().catch((err) => console.warn("Failed to suspend AudioContext on mute:", err));
  } else if (!muted && audioCtx) {
    audioCtx.resume().catch((err) => console.warn("Failed to resume AudioContext on unmute:", err));
  }
};

export const getMuteState = (): boolean => {
  return isMuted;
};

const playSynthTone = (
  frequencies: number[],
  duration: number,
  type: OscillatorType = "sine",
  volume = 0.1,
  frequencySweep?: { endFreq: number; duration: number }
) => {
  const ctx = initAudio();
  if (!ctx || ctx.state === "suspended" || isMuted) return;

  try {
    const time = ctx.currentTime;
    const osc = ctx.createOscillator();
    const gainNode = ctx.createGain();

    osc.type = type;
    
    // Set starting frequency
    if (frequencies.length > 0) {
      osc.frequency.setValueAtTime(frequencies[0], time);
      // If there are multiple tones (arpeggio), schedule them
      frequencies.forEach((freq, idx) => {
        if (idx > 0) {
          osc.frequency.setValueAtTime(freq, time + (idx * duration) / frequencies.length);
        }
      });
    }

    // Apply frequency sweep
    if (frequencySweep && frequencies.length > 0) {
      osc.frequency.exponentialRampToValueAtTime(
        frequencySweep.endFreq,
        time + frequencySweep.duration
      );
    }

    // Gain envelope
    gainNode.gain.setValueAtTime(volume, time);
    gainNode.gain.exponentialRampToValueAtTime(0.001, time + duration);

    osc.connect(gainNode);
    gainNode.connect(ctx.destination);

    osc.start(time);
    osc.stop(time + duration);
  } catch (err) {
    console.warn("Failed to play synth sound:", err);
  }
};

# ORBI ENERGY SWARM (v0.1.0-prototype-recovery-baseline)

A standalone hardened React, TypeScript, and HTML5 Canvas game. This release represents a frozen visual and technical baseline recovered from the ORBI Life game ecosystem.

## Core Objective
Pilot the **Orbi Foton** leader, synchronize with an autonomous tactical swarm of energy collectors, and harvest crystallized energy fragments while repelling hostile parasites and drones across six procedurally shifting cybernetic biomes.

## Architecture
- **Framework**: React 19 + TypeScript + Tailwind CSS (v4)
- **Engine**: Single unified `requestAnimationFrame` game loop with `deltaTime` physics.
- **Rendering**: HTML5 Canvas for high-frequency objects; React coordinates low-frequency HUD overlays and sidebar dashboards.
- **Audio**: Custom synthesized procedural Web Audio API manager.
- **State**: Safe local storage saving for high-scores, progression credits, active mementos, and audio settings.

## Getting Started
```bash
# Install dependencies

# Run the dev server
npm run dev

# Build for production

```

# Technical Fixes & Hardening Report

This document records the engineering fixes implemented to stabilize the ORBI ENERGY SWARM visual prototype into a fully performant, robust standalone application.

## 1. Single-Loop Unified Engine (`requestAnimationFrame`)
- **Issue**: The original baseline was prone to multiplying run loops upon React rendering triggers or state changes.
- **Fix**: Designed a single stable `requestAnimationFrame` mounted within a unified React `useEffect` cycle. All positions, physics states, key states, and particle frames reside within persistent `useRef` buffers. React state modifications are throttled to 10-frame intervals and applied only for low-frequency HUD displays.

## 2. Reusable Web Audio procedural Synthesizer
- **Issue**: Standard audio implementations create new instances of `AudioContext` on every action, triggering browser overhead crashes.
- **Fix**: Programmed a lazy-initialized singleton `AudioContext` that remains suspended until the user executes their first core connection trigger. The class is recycled, supports global mute-state checks, and disposes correctly upon component unmount.

## 3. Double Collision Prevention & Damage Control
- **Issue**: High-frequency collision bounds checking caused multiple projectile hits to apply to the same entity on consecutive frames, triggering duplicated scoring, immediate player deaths, or immortal entities.
- **Fix**: 
  - Enemies and player projectiles are marked with explicit flags (`isDead` or lifetime expire triggers) immediately upon detecting the initial contact. Dead entities are instantly ignored in subsequent collision sweeps.
  - Added an invincibility frame cooldown (`playerRef.current.invincibilityTimer`) of 1000ms to the player, visually indicated by high-contrast flashing.

## 4. Entity Cap Constraints
- **Issue**: Continuous spawning of bullets, explosion sparks, or enemies threatened browser memory leaks.
- **Fix**: Defined strict caps (`MAX_SWARM_MEMBERS`, `MAX_ENEMIES`, `MAX_PROJECTILES`, `MAX_PARTICLES`, `MAX_RESOURCES`). Spawning mechanisms reject new instances if boundaries are exceeded, keeping operations clean and smooth.

## 5. Input Hardening
- **Issue**: Arrow keys triggered browser page scrolls, and key stuck bugs occurred when tabs lost focus.
- **Fix**:
  - Implemented `e.preventDefault()` on gaming-specific keys (Arrows, Space).
  - Attached listeners to `blur` and `visibilitychange` to automatically flush all active pressed key buffers.
